# S2.6-R5 report — the blur excludes the player hull, shader-side (AC7)

## Result

and reads the `&"player_ship"` group) pushes the hull's own screen disc — `exclude_player`,
`player_screen_uv`, `player_screen_radius` — and `speed_blur.gdshader` skips **the smear and

The region: the hull's own radius as CONTRACTS §14's F16 pins it for a hull
(`CollisionShape2D.radius`, **30 u** on the shipped `player_ship.tscn` — read by type, a
`CircleShape2D` under the ship's `RigidBody2D`, so no node name is hard-coded), scaled by the
camera's applied zoom and divided by the viewport's visible size. That last division makes the
figure invariant across window sizes and the `canvas_items` stretch scale (a pure scale cancels
in UV). No fallback number was invented: a scene with no hull in the group, or a hull whose
collider carries no circle, pushes `exclude_player = false` **and** a zero radius, and the blur
stays full-screen.

```gdscript
extends SceneTree
## S2.6-R5's on-GPU pixel measurement of the blur's hull exclusion (AC7).
const Speed := preload("res://game/speed_fantasy.gd")

const SIZE := Vector2i(640, 360)
const RADIUS := 30.0
const TAG := "[R5P]"

var _view: SubViewport = null
var _fantasy: Node = null
var _camera: Camera2D = null


func _initialize() -> void:
	_build()
	await process_frame
	await process_frame
	_fantasy.call(&"set_wheel_zoom", 1.0)
	_fantasy.call(&"set_ratio", 1.0, Vector2(400.0, 0.0))
	var material := (_fantasy.call(&"blur_rect") as ColorRect).material as ShaderMaterial
	## The dust emitter is the only other thing on this layer: hidden, so the two readings
	## differ by the blur alone (the applied zoom is 0.82 in both - the reference must not be
	## taken at ratio 0, where the pull-back puts the zoom back to 1.0).
	var dust := _fantasy.call(&"dust") as GPUParticles2D
	dust.emitting = false
	dust.visible = false
	var excluded := await _frame()
	material.set_shader_parameter(&"blur_strength", 0.0)
	var clean := await _frame()
	material.set_shader_parameter(&"blur_strength", float(_fantasy.call(&"blur_strength")))
	material.set_shader_parameter(&"exclude_player", false)
	var control := await _frame()
	material.set_shader_parameter(&"exclude_player", true)

	var radius_px := float(_fantasy.call(&"hull_screen_radius")[0]) * float(SIZE.x)
	var centre := (_fantasy.call(&"hull_screen_uv") as Vector2) * Vector2(SIZE)
	var rule := centre + Vector2(radius_px * 3.0, 0.0)
	print(
		(
			"%s HEAD size=%s zoom=%.4f radius_units=%.2f radius_px=%.2f span_px=%.2f "
			+ "aberration_px=%.2f uv=%s"
		)
		% [
			TAG, str(SIZE), float(_fantasy.call(&"applied_zoom")),
			float(_fantasy.call(&"hull_radius_units")), radius_px,
			Speed.BLUR_SPAN_PX * float(_fantasy.call(&"blur_strength")),
			Speed.ABERRATION_PX * float(_fantasy.call(&"chromatic_aberration")),
			str(_fantasy.call(&"hull_screen_uv")),
		]
	)
	_row("HULL_EXCLUDED", _disc(excluded, clean, centre, radius_px))
	_row("BACKGROUND", _box(excluded, clean, rule, radius_px * 1.2))
	_row("HULL_CONTROL", _disc(control, clean, centre, radius_px))
	var band := int(centre.y + radius_px * 2.0)
	print(
		"%s SMEAR rule_row lit_clean=%d lit_excluded=%d span_clean=%d span_excluded=%d"
		% [
			TAG, _lit(clean, band), _lit(excluded, band),
			_span(clean, band), _span(excluded, band),
		]
	)
	print(
		"%s SPLIT hull_clean=%.3f hull_excluded=%.3f hull_control=%.3f back_clean=%.3f back_excluded=%.3f"
		% [
			TAG, _split(clean, centre), _split(excluded, centre), _split(control, centre),
			_split(clean, centre + Vector2(120.0, -60.0) * 0.82),
			_split(excluded, centre + Vector2(120.0, -60.0) * 0.82),
		]
	)
	quit()


func _row(label: String, read: Dictionary) -> void:
	print(
		"%s %s px=%d max=%.6f mean=%.6f"
		% [TAG, label, int(read[&"count"]), read[&"max"], read[&"mean"]]
	)


func _build() -> void:
	_view = SubViewport.new()
	_view.size = SIZE
	_view.render_target_update_mode = SubViewport.UPDATE_ALWAYS
	root.add_child(_view)
	var hull := Node2D.new()
	_view.add_child(hull)
	hull.add_to_group(Speed.PLAYER_GROUP, true)
	var body := RigidBody2D.new()
	hull.add_child(body)
	var shape := CollisionShape2D.new()
	var circle := CircleShape2D.new()
	circle.radius = RADIUS
	shape.shape = circle
	body.add_child(shape)
	_view.add_child(Painter.new())
	_camera = Camera2D.new()
	_view.add_child(_camera)
	_fantasy = Speed.new()
	_view.add_child(_fantasy)
	_fantasy.call(&"bind_camera", _camera)


func _frame() -> Image:
	await RenderingServer.frame_post_draw
	await RenderingServer.frame_post_draw
	return _view.get_texture().get_image()


func _disc(blurred: Image, clean: Image, centre: Vector2, radius_px: float) -> Dictionary:
	return _read(blurred, clean, centre, radius_px * 1.7, true)


func _box(blurred: Image, clean: Image, centre: Vector2, half: float) -> Dictionary:
	return _read(blurred, clean, centre, half, false)


func _read(blurred: Image, clean: Image, centre: Vector2, half: float, round: bool) -> Dictionary:
	var count := 0
	var max_delta := 0.0
	var total := 0.0
	for y in range(int(centre.y - half), int(centre.y + half) + 1):
		for x in range(int(centre.x - half), int(centre.x + half) + 1):
			if x < 0 or y < 0 or x >= SIZE.x or y >= SIZE.y:
				continue
			if round and Vector2(x + 0.5, y + 0.5).distance_to(centre) > half * 0.5:
				continue
			var got := blurred.get_pixel(x, y)
			var want := clean.get_pixel(x, y)
			var delta := maxf(absf(got.r - want.r), maxf(absf(got.g - want.g), absf(got.b - want.b)))
			max_delta = maxf(max_delta, delta)
			total += delta
			count += 1
	return {&"count": count, &"max": max_delta, &"mean": total / maxf(float(count), 1.0)}


func _lit(image: Image, row: int) -> int:
	var count := 0
	for x in SIZE.x:
		var colour := image.get_pixel(x, row)
		if maxf(colour.r, maxf(colour.g, colour.b)) > 0.02:
			count += 1
	return count


func _span(image: Image, row: int) -> int:
	var first := -1
	var last := -1
	for x in SIZE.x:
		var colour := image.get_pixel(x, row)
		if maxf(colour.r, maxf(colour.g, colour.b)) > 0.02:
			if first < 0:
				first = x
			last = x
	return (last - first + 1) if first >= 0 else 0


## The chromatic split as a mean vertical offset between the red and the blue profile of a
## small bright square (the shader offsets the channels across the smear's axis).
func _split(image: Image, spot: Vector2) -> float:
	var red_weight := 0.0
	var red_total := 0.0
	var blue_weight := 0.0
	var blue_total := 0.0
	for x in range(int(spot.x - 3), int(spot.x + 4)):
		for y in range(int(spot.y - 10), int(spot.y + 11)):
			if x < 0 or y < 0 or x >= SIZE.x or y >= SIZE.y:
				continue
			var colour := image.get_pixel(x, y)
			red_weight += colour.r * float(y)
			red_total += colour.r
			blue_weight += colour.b * float(y)
			blue_total += colour.b
	if red_total <= 0.0 or blue_total <= 0.0:
		return 0.0
	return (red_weight / red_total) - (blue_weight / blue_total)


class Painter extends Node2D:
	func _draw() -> void:
		draw_rect(Rect2(-4000.0, -4000.0, 8000.0, 8000.0), Color.BLACK)
		## The hull: a striped disc of the collider's own radius, so a smear washes it out.
		for i in int(RADIUS * 2.0):
			var x := -RADIUS + float(i)
			var half := sqrt(maxf(RADIUS * RADIUS - x * x, 0.0))
			draw_rect(Rect2(x, -half, 1.0, half * 2.0), Color.WHITE if i % 4 < 2 else Color.BLACK)
		draw_rect(Rect2(-1.5, -1.5, 3.0, 3.0), Color.WHITE)
		## The background rule the smear spreads into a band, and the square the split reads.
		draw_rect(Rect2(90.0, -300.0, 2.0, 600.0), Color.WHITE)
		draw_rect(Rect2(118.5, -61.5, 3.0, 3.0), Color.WHITE)
```

**Two facts that shaped the implementation, both measured:**

- Godot's fragment processor rejects `return` ("Using 'return' in the 'fragment' processor
  function is incorrect", surfaced as `ERROR: Shader compilation failed.` at
  `shader_set_code`). The guard is therefore one condition on the existing branch, not an early
  out. A shader that fails to compile also **loses its uniform list**, which is what the suite's
  first test asserts against (`get_shader_uniform_list()` reports `exclude_player` TYPE_BOOL and
  the two TYPE_VECTOR2 uniforms, so a broken shader fails the gate rather than only printing).
- The region first shipped one frame stale: `_apply()` ran the blur before the zoom, so the disc
  was computed from the previous frame's `Camera2D.zoom` (measured: 30.00 px where the applied
  zoom 0.82 gives 24.60 px). Cured by applying the zoom before the blur; `speed_fantasy.gd` is the
  camera's single writer, so nothing else reorders.

## Deviations

1. **AC7's pixel half is measured by a probe that is not in the tree — the dispatch's call, flagged
   for the orchestrator (bucket 1, code route; no pause).** The dispatch says "Add ONLY
   tests/test_s2_6_blur.gd", so the probe was deleted after measuring and its full source is
   embedded above; the acceptance is reproduced by copying that block in. **R2's own
   `probe_s2_6_burst.gd` is in the tree**, so if the orchestrator prefers the wave's probe
   convention over the literal "ONLY", this is a one-file addition that R6 can re-run
   byte-identically; say the word and it goes back as `tests/probe_s2_6_blur.gd`. The gate itself
   cannot carry this measurement (Deviation 3).
2. **The suite's AC7 comparison is an in-process evaluation of the shipped fragment
   (`_fragment_pixel`), not a framebuffer read.** It is the shipped algorithm — the pass-through
   tap, the region test, the 17-tap triangular smear and the channel split — evaluated over a
   synthetic framebuffer built around the disc the shipped material carries, with the shipped
   uniforms and pixel spans read from the material. Reversal: delete the evaluation and keep the
   structural pins. What keeps it honest: the structural test pins the shipped source's guard
   order and both effects inside it, the tap count it evaluates against is the shader's own
   (`const int TAPS`), and every number it uses comes out of the material the wiring pushed.
3. **No in-gate framebuffer read exists, and no awaiting test can be leaned on.** Two measured
   facts, both worth a LOW row (below): `force_draw()` does not rebuild draw commands, and the
   runner scores an awaiting test method **PASS immediately** — a temporary method
   `await _host().get_tree().process_frame; assert_true(false, ...)` was reported
   `[PASS] ... test_temporary_awaiting_probe`, `[SUMMARY] passed=3 failed=0` (the continuation
   never runs, because the runner quits at the end of `_ready`). Any future in-suite frame
   stepping must fix the runner first, or it silently passes.
4. **`_apply()`'s three calls are reordered** (zoom, dust, blur). Not a taste change: the region
   must be derived from this frame's applied zoom, and the value it read was one frame stale
   (measured above). Reversal: the old order plus a one-frame-stale region.
5. **No pinned number moved.** `max_speed`, the `blur_strength`/`pullback` bands, `blur_span_px`,
   `aberration_px`, the dust and vignette rows, blend modes, sheets and layers: none touched. The
   region's own two figures are derived (the hull's collider radius, the camera's applied zoom),
   and the one number this pass introduces is the boolean `SPEED_BLUR_EXCLUDE_PLAYER`, which
   CONTRACTS §14 pins.
6. **Hook workaround (evidence for L92).** `.crush/hooks/enforce_worker_files.py`'s `_norm` strips
   only the Windows workspace root, so every absolute Linux path is denied as "outside this
   worker's declared file set" — including `.agents/` and `vajb-orbit/tests/`. The first edit of
   this pass was blocked; all edits thereafter used workspace-relative `file_path` values, which
   the hook compares correctly. Same defect R1 recorded.
7. Not mine, seen in the tree: `RCLONE_TEST` shows as deleted, and this pass touched no frozen
   file (`vajb-orbit/project.godot`, `docs/gameplay/18_engine_spec.md`, `assets/`, `addons/`).
